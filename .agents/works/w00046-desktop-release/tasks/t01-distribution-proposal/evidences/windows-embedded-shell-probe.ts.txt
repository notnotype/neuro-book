// 在 Windows 上试 pi-natives 的内嵌 shell：Agent 的 shell 工具就是这么调的（nb-harness kit/tools/shell.ts）。
import {executeShell} from "@oh-my-pi/pi-natives";
import {mkdtempSync, writeFileSync} from "node:fs";
import {tmpdir} from "node:os";
import {join} from "node:path";

const dir = mkdtempSync(join(tmpdir(), "nb-shell-"));
writeFileSync(join(dir, "第一章.md"), "林青走进药铺。\n周叔抬起头。\n");
writeFileSync(join(dir, "b.txt"), "b\na\nc\n");
const cases: Array<[string, string]> = [
    ["echo 与变量", 'X=世界; echo "你好 $X"'],
    ["ls 与中文文件名", "ls"],
    ["cat 与管道 wc", "cat 第一章.md | wc -l"],
    ["sort uniq head", "sort b.txt | uniq | head -2"],
    ["sed", "sed 's/周叔/周伯/' 第一章.md"],
    ["find", "find . -name '*.md'"],
    ["jq", "echo '{\"a\":[1,2]}' | jq '.a | length'"],
    ["for 循环与 $(...)", 'for f in *.txt; do echo "$f:$(wc -l < $f)"; done'],
    ["重定向写文件", "echo 新 > c.txt && cat c.txt"],
    ["mkdir cp mv rm", "mkdir d && cp b.txt d/ && mv d/b.txt d/e.txt && ls d && rm -r d && echo ok"],
    ["git（系统的）", "git --version"],
    ["awk（不内置）", "echo a b | awk '{print $2}'"],
    ["python（不内置）", "python --version"],
    ["Windows 程序 where", "where.exe cmd"],
    ["退出码", "exit 3"],
];
for (const [name, command] of cases) {
    let out = "";
    const started = performance.now();
    try {
        const result = await executeShell({command, cwd: dir, timeoutMs: 15_000}, (_error, chunk) => {
            out += chunk;
        });
        console.log(`## ${name}\n$ ${command}\nexit=${String(result.exitCode)} timedOut=${result.timedOut} ${(performance.now() - started).toFixed(0)}ms\n${out.trimEnd()}\n`);
    } catch (error) {
        console.log(`## ${name}\n$ ${command}\nthrew: ${String(error)}\n`);
    }
}
