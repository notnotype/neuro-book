import {afterAll, afterEach, beforeAll, expect, test} from "bun:test";
import {rm, readFile, mkdtemp} from "node:fs/promises";
import {join} from "node:path";
import {tmpdir} from "node:os";

const A = "/home/<user>/CodeRepository/neuro-book/packages/neuro-book/src/plugins/agent";
const {MAIN_PROFILE} = await import(`${A}/shared/contracts`);
const {AGENT_MODES_EXTENSION} = await import(`${A}/shared/permissions`);
const {fakeEndpoint, localModelSettings} = await import(`${A}/testing/endpoint`);
const {agentWorld} = await import(`${A}/testing/world`);

let root = "";
let endpoint: any;
const worlds: any[] = [];
beforeAll(async () => { root = await mkdtemp(join(tmpdir(), "rv-approval-r2-")); endpoint = fakeEndpoint(); });
afterEach(async () => { endpoint.reset(); for (const w of worlds.splice(0)) await w.close(); });
afterAll(async () => { endpoint.stop(); if (root !== "") await rm(root, {recursive: true, force: true}); });
const cid = () => crypto.randomUUID();
let n = 0;
async function open(mode: string, approval: string, extra: Record<string, unknown> = {}) {
    n += 1;
    const w = await agentWorld(join(root, `w${n}`), {userSettings: {...localModelSettings(endpoint.baseUrl), ...extra}});
    worlds.push(w);
    const created = await w.service.create({profile: MAIN_PROFILE});
    if (!created.ok) throw new Error(created.code);
    const changed = await created.value.changeSettings({commandId: cid(), patch: {core: {approvalMode: approval}, [AGENT_MODES_EXTENSION]: {mode}}});
    expect(changed.ok).toBe(true);
    return {session: created.value, w};
}
async function outcome(s: any, tool: string, args: any) {
    const before = (await s.path()).length;
    endpoint.script({tool, args}, "好");
    await s.send({commandId: cid(), input: {text: "做"}});
    for (let i = 0; i < 200; i += 1) {
        const ev = (await s.path()).slice(before);
        const asked = ev.find((e: any) => e.type === "interactionRequested");
        if (asked) return {asked: asked.data.request};
        const res = ev.find((e: any) => e.type === "toolResult");
        if (res) return {result: {status: res.data.status, code: res.data.code, text: String(res.data.text).slice(0, 200)}, started: ev.find((e: any) => e.type === "toolStarted")?.data.approval};
        await new Promise((r) => setTimeout(r, 25));
    }
    return {timeout: true};
}

test("A plan+ask, no project: cd elsewhere then relative redirect", async () => {
    const {session} = await open("plan", "ask");
    const out = await outcome(session, "shell", {command: "cd /tmp/review-approval-r2/outside && echo pwned > x.txt"});
    console.log("A ->", JSON.stringify(out));
    console.log("A file:", await readFile("/tmp/review-approval-r2/outside/x.txt", "utf8").catch((e) => String(e.code)));
});

test("B normal+ask: write tool to host-written dir and to another session", async () => {
    const {session} = await open("normal", "ask");
    console.log("B1 ->", JSON.stringify(await outcome(session, "write", {path: "session://attachments/a.txt", content: "x"})));
});

test("C normal+ask: user ask rule then appended allow (what 以后都允许 writes)", async () => {
    const {session} = await open("normal", "ask", {"nbook.agent/permissionRules": [{tool: "write", action: "ask", reason: "我要看每次写"}, {tool: "write", action: "allow"}]});
    console.log("C ->", JSON.stringify(await outcome(session, "write", {path: "session://a.md", content: "x"})));
});
