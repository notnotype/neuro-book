// t14 计划审查：第三轮 11 条与计划依赖的时序，按现有内核跑一次（只读仓库）
import {test} from "bun:test";
const K = "/home/notnotype/CodeRepository/neuro-book/packages/nb-harness/src/kernel";
const {createTestSession} = await import(`${K}/testing/test-session.ts`);
const {defineExtension} = await import(`${K}/extension/define.ts`);
const {notifyAssemblyChanged, closeSession} = await import(`${K}/session/session.ts`);

let n = 0;
const id = () => `rv-${++n}`;
const gate = () => {
    let open!: () => void;
    const p = new Promise<void>((r) => (open = r));
    return {p, open};
};
const parent = {kind: "session", id: "parent"} as const;
const bufferOf = async (s: any) => (await s.raw.snapshot()).buffer.map((b: any) => [b.mode, b.input.text, b.by?.kind ?? "user"]);
const allEvents = async (s: any) => (await s.raw.log({afterSeq: 0, limit: 2000})).events;
const typesOf = async (s: any, keep: string[]) => (await allEvents(s)).filter((e: any) => keep.includes(e.type)).map((e: any) => e.type + (e.type === "runEnded" ? `(${e.data.reason})` : "")).join(" ");

test("R1 yield 那一轮里到达的会话插话", async () => {
    const entered = gate();
    const hold = gate();
    const ext = defineExtension({id: "lab.gate", on: {assistant: async () => { entered.open(); await hold.p; }}});
    const schema = {type: "object", properties: {title: {type: "string"}}, required: ["title"]};
    await using s = await createTestSession({extensions: [ext], result: {mode: "optional", schema}});
    s.model.script({tools: [{tool: "yield", args: {title: "雨夜"}}]});
    const sent = await s.raw.send({commandId: id(), input: {text: "起个标题"}});
    if (!sent.ok) throw new Error("send failed");
    await entered.p;
    const steer = s.raw.as(parent).send({commandId: id(), input: {text: "改成两个字以内"}, mode: "steer"});
    hold.open();
    const r = await steer;
    const result = await s.raw.waitRun(sent.value.runId!);
    console.log("R1 steer send:", JSON.stringify(r.ok ? r.value : r), "first run:", sent.value.runId, "reason:", result.reason, "buffer:", JSON.stringify(await bufferOf(s)));
});

test("R2a 会话身份在用户运行暂停的叶子上 send：接着跑用户的运行", async () => {
    await using s = await createTestSession({});
    s.model.script("答一", "答二", "新答二");
    await s.raw.enqueue({commandId: id(), input: {text: "第二问"}, mode: "followup"});
    await s.say("第一问");
    const before = (await s.raw.rewindTargets()).find((t: any) => t.kind === "before-turn" && t.inputText === "第二问");
    await s.raw.rewind({commandId: id(), to: before.id});
    const runIdBefore = (await allEvents(s)).find((e: any) => e.type === "runStarted").data.runId;
    const sent = await s.raw.as(parent).send({commandId: id(), input: {text: "会话的话"}});
    console.log("R2a user run:", runIdBefore, "session send:", JSON.stringify(sent.ok ? sent.value : sent));
    if (sent.ok && sent.value.runId) console.log("R2a resumed:", (await s.raw.waitRun(sent.value.runId)).reason, "runStarted count:", (await allEvents(s)).filter((e: any) => e.type === "runStarted").length);
});

test("R8 用户缺省停止进行中，会话发的话被一起清掉", async () => {
    const entered = gate();
    let finishTool!: () => void;
    const slow = {id: "slow", execute: (_a: any, ctx: any) => new Promise<string>((resolve) => { entered.open(); ctx.signal.addEventListener("abort", () => { finishTool = () => resolve("stopped"); }); })};

test("N1 运行中撤回扩展；收口批反应期间到达的 send 排在重新装配的队列项之前", async () => {
    const entered = gate();
    const hold = gate();
    const toolIn = gate();
    const toolHold = gate();
    const endgate = defineExtension({id: "lab.endgate", on: {runEnded: async () => { entered.open(); await hold.p; }}});
    const victim = defineExtension({id: "lab.victim", on: {assistant: async () => {}}});
    await using s = await createTestSession({extensions: [endgate, victim], tools: [{id: "slow", execute: async () => { toolIn.open(); await toolHold.p; return "ok"; }}]});
    s.model.script({tools: [{tool: "slow"}]}, "答一", "答二");
    const sent = await s.raw.send({commandId: id(), input: {text: "第一句"}});
    if (!sent.ok) throw new Error("send failed");
    await toolIn.p;
    const fakeEnv = {profiles: {resolve: () => undefined}, extensions: () => [{extension: endgate, plugin: null}]};
    const revoked = notifyAssemblyChanged(s.raw, fakeEnv as any, {removed: ["lab.victim"]});
    toolHold.open();
    await entered.p;
    const late = s.raw.send({commandId: id(), input: {text: "反应期间到达的 send"}});
    hold.open();
    const r = await late;
    await revoked;
    console.log("N1 late send:", JSON.stringify(r.ok ? r.value : r));
    console.log("N1 order:", await typesOf(s, ["runEnded", "inputQueued", "profileChanged", "runStarted"]));
    console.log("N1 snapshot pendingReassembly:", JSON.stringify((await s.raw.snapshot()).pendingReassembly ?? null), "status:", (await s.raw.snapshot()).status.state);
});

test("N3 受理批的反应进行中开始关闭：受理项写完后仍 launch 新运行，关闭不等它", async () => {

    s.model.script("答一");

    console.log("N3 model requests:", s.model.requests.length);
});
