== 脚本
// 语法色缺省推导的候选：对每套自带与护眼配色，量各变量在 --bg-main 上的对比度与两两色差（CIE76）。
const R = "/home/<user>/CodeRepository/neuro-book/packages";
const {nbookColorways} = await import(`${R}/nb-ui/themes/nbook/colorways.ts`);
const {macosColorways} = await import(`${R}/nb-ui/themes/macos/colorways.ts`);
const eye = await import(`${R}/neuro-book/src/plugins/workbench/shared/eye-care-themes.ts`);
type V = Record<string, string>;
const looks: Record<string, V> = {...(nbookColorways as Record<string, V>), ...(macosColorways as Record<string, V>)};
const base = nbookColorways as Record<string, V>;
for (const [name, set] of [["eye-green", eye.EYE_GREEN_LOOKS], ["eye-sepia", eye.EYE_SEPIA_LOOKS]] as const) {
    for (const mode of ["light", "dark"] as const) looks[`${name}-${mode}`] = {...base[`nbook-${mode}`]!, ...((set as Record<string, {colors?: V}>)[mode]?.colors ?? {})};
}
const rgb = (hex: string): number[] => [1, 3, 5].map((i) => parseInt(hex.slice(i, i + 2), 16));
const hexOf = (c: number[]): string => `#${c.map((v) => Math.round(v).toString(16).padStart(2, "0")).join("")}`;
const mix = (a: string, pa: number, b: string): string => hexOf(rgb(a).map((v, i) => v * pa + rgb(b)[i]! * (1 - pa)));
const lum = (hex: string): number => { const v = rgb(hex).map((c) => c / 255).map((c) => (c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4)); return 0.2126 * v[0]! + 0.7152 * v[1]! + 0.0722 * v[2]!; };
const ratio = (a: string, b: string): number => { const [x, y] = [lum(a), lum(b)].sort((p, q) => q - p); return (x! + 0.05) / (y! + 0.05); };
const lab = (hex: string): number[] => { const [r, g, b] = rgb(hex).map((c) => c / 255).map((c) => (c <= 0.04045 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4)); const x = (0.4124 * r! + 0.3576 * g! + 0.1805 * b!) / 0.95047, y = 0.2126 * r! + 0.7152 * g! + 0.0722 * b!, z = (0.0193 * r! + 0.1192 * g! + 0.9505 * b!) / 1.08883; const f = (t: number) => (t > 0.008856 ? Math.cbrt(t) : 7.787 * t + 16 / 116); return [116 * f(y) - 16, 500 * (f(x) - f(y)), 200 * (f(y) - f(z))]; };
const dE = (a: string, b: string): number => Math.hypot(...lab(a).map((v, i) => v - lab(b)[i]!));
const S = Number(process.env.S ?? "0.8");
const strong = (v: V, c: string): string => mix(c, S, v["--text-main"]!);
const palette = (v: V): Record<string, string> => ({
    comment: mix(v["--text-secondary"]!, 0.5, v["--text-muted"]!),
    keyword: strong(v, mix(v["--accent-text"]!, 0.55, v["--status-danger"]!)),
    string: strong(v, v["--status-success"]!),
    number: strong(v, v["--status-warning"]!),
    type: strong(v, v["--status-info"]!),
    tag: strong(v, v["--status-danger"]!),
    attribute: strong(v, mix(v["--status-info"]!, 0.5, v["--status-success"]!)),
    regexp: strong(v, mix(v["--status-danger"]!, 0.5, v["--status-warning"]!)),
    variable: v["--text-main"]!,
    punctuation: v["--text-secondary"]!,
});
const pairs = [["attribute", "string"], ["attribute", "number"], ["attribute", "keyword"], ["tag", "attribute"], ["tag", "string"], ["tag", "keyword"], ["keyword", "type"], ["keyword", "string"], ["keyword", "number"], ["regexp", "string"], ["regexp", "number"], ["type", "attribute"], ["type", "string"], ["tag", "regexp"], ["tag", "number"], ["keyword", "tag"], ["keyword", "regexp"], ["attribute", "type"], ["number", "string"]];
for (const [name, v] of Object.entries(looks)) {
    if (!v["--bg-main"]?.startsWith("#")) { console.log(name, "bg-main 不是十六进制", v["--bg-main"]); continue; }
    const p = palette(v);
    const low = Object.entries(p).filter(([k, c]) => ratio(c, v["--bg-main"]!) < (k === "comment" ? 3 : 4.5)).map(([k, c]) => `${k} ${ratio(c, v["--bg-main"]!).toFixed(2)}`);
    const near = pairs.map(([a, b]) => [a, b, dE(p[a!]!, p[b!]!)] as const).filter(([, , d]) => d < 15).map(([a, b, d]) => `${a}~${b} ${d.toFixed(1)}`);
    console.log(`${name.padEnd(18)} ${Object.values(p).join(" ")}\n   bg ${v["--bg-main"]}  不达标: ${low.join(", ") || "无"}  近色(<15): ${near.join(", ") || "无"}`);
}
