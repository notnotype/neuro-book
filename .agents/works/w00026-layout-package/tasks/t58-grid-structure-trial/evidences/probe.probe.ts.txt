import {describe, expect, it} from "bun:test";
import {
    GRID_MAX_DEPTH,
    GRID_SNAPSHOT_VERSION,
    createGrid,
    type GridNode,
    type GridNodeInput,

const shape = (node: GridNode<unknown> | null): unknown =>
    node === null ? null : node.kind === "leaf" ? `${node.id}(${node.sizing}:${node.size.width}x${node.size.height})` : {[`${node.id}[${node.orientation[0]}]`]: node.children.map(shape)};
const depthOf = (node: GridNode<unknown> | null): number => (node === null ? 0 : node.kind === "leaf" ? 1 : 1 + Math.max(0, ...node.children.map(depthOf)));
const widths = (sizes: Record<string, {width: number; height: number}>, ids: string[]) => Object.fromEntries(ids.map((id) => [id, Math.round(sizes[id]!.width * 100) / 100]));
const log = (label: string, value: unknown) => console.log(`## ${label}\n${JSON.stringify(value, null, 1)}`);

describe("P1 同向 splitLeaf", () => {
    it("横排里横着拆：造出嵌套分支；反复拆最后一块，几次到深度上限", () => {
        const grid = createGrid<string>({kind: "branch", id: "root", orientation: "horizontal", children: [
            {kind: "leaf", id: "a", ref: "a", size: {width: 1, height: 0}},
            {kind: "leaf", id: "b", ref: "b", size: {width: 1, height: 0}},
        ]}, {sashSize: 4});
        const r = grid.splitLeaf("b", {branchId: "s1", orientation: "horizontal", side: "after", leaf: {kind: "leaf", id: "c", ref: "c"}});
        log("P1a 一次同向拆分后的树", {result: r, tree: shape(grid.root())});
        log("P1a 呈现（1204 宽）", widths(grid.layout({width: 1204, height: 600}).sizes, ["a", "s1", "b", "c"]));
        let last = "c";
        let n = 1;
        for (;;) {
            const next = `x${n}`;
            const res = grid.splitLeaf(last, {branchId: `s${n + 1}`, orientation: "horizontal", side: "after", leaf: {kind: "leaf", id: next, ref: next}});
            if (!res.ok) {
                log("P1b 连续同向拆最后一块：第几次被拒、拒绝内容、此时树深", {attempt: n + 1, res, depth: depthOf(grid.root())});
                break;
            }
            last = next;
            n += 1;
        }
    });
});

describe("P3 深度计法（第 20 条）", () => {
    // 一条 n 层的链：根是第 1 层，叶在第 n 层。
    const chain = (levels: number): GridNodeInput<string> => {
        let node: GridNodeInput<string> = {kind: "leaf", id: "end", ref: "end", size: {width: 1, height: 1}};
        for (let level = levels - 1; level >= 1; level -= 1) node = {kind: "branch", id: `l${level}`, orientation: level % 2 ? "horizontal" : "vertical", size: {width: 1, height: 1}, children: [node, {kind: "leaf", id: `side${level}`, ref: `side${level}`, size: {width: 1, height: 1}}]};
        return node;
    };
    it("restore 允许几层、恢复后能不能拆；createGrid 建树查不查深度", () => {
        for (const levels of [16, 17, 18]) {
            const source = createGrid<string>(chain(levels));
            const snapshot = source.serialize();
            const grid = createGrid<string>(null);
            const restored = grid.restore(JSON.parse(JSON.stringify(snapshot)), (ref) => ({ref}));
            const addShallow = restored.ok ? grid.addLeaf("l1", 0, {kind: "leaf", id: "new", ref: "new"}) : null;
            const removeShallow = restored.ok ? createGrid<string>(chain(levels)).removeLeaf("side1") : null;
            log(`P3 ${levels} 层`, {
                treeDepth: depthOf(source.root()),
                restore: restored.ok ? "ok" : {code: restored.code, message: restored.message},
                addLeafAtRootAfterRestore: addShallow,
                createGridThenAddAtRoot: createGrid<string>(chain(levels)).addLeaf("l1", 0, {kind: "leaf", id: "new", ref: "new"}),
                createGridThenRemove: removeShallow,
            });
        }
        expect(GRID_MAX_DEPTH).toBe(16);
    });
});

describe("P5 serialize/restore 带不带策略", () => {
    it("快照里没有 sizing、约束、collapse；新网格恢复用解析结果，原地恢复回落到当前树", () => {
        const input: GridNodeInput<string> = {kind: "branch", id: "root", orientation: "horizontal", children: [
            {kind: "leaf", id: "side", ref: "side", size: {width: 240, height: 0}, sizing: "fixed", minimumSize: {width: 160, height: 0},
                collapse: {collapsed: true, collapsedSize: 0, restoreSize: 300, collapseThreshold: 24, expandThreshold: 24}},
            {kind: "leaf", id: "main", ref: "main", size: {width: 1, height: 0}},
        ]};
        const grid = createGrid<string>(input, {sashSize: 4});
        const snapshot = grid.serialize();
        log("P5a 快照", snapshot);
        const fresh = createGrid<string>(null, {sashSize: 4});
        fresh.restore(snapshot, (ref) => ({ref}));
        log("P5b 新网格恢复（解析只给 ref）后的 side", fresh.find("side"));
        const inPlace = createGrid<string>(input, {sashSize: 4});
        inPlace.restore(snapshot, (ref) => ({ref}));
        log("P5c 原地恢复（解析只给 ref）后的 side：回落到当前树", inPlace.find("side"));
        const ignored = createGrid<string>(null, {sashSize: 4});
        const bad = ignored.restore({version: GRID_SNAPSHOT_VERSION, root: {kind: "leaf", id: "x", ref: 5, size: {width: 0, height: 0}}}, (ref) => ({ref}));
        log("P5d 恢复失败而宿主没看结果：树仍是 null", {ok: bad.ok, root: ignored.root()});
        // sashSize 不在快照里：新网格不给 sashSize 时缝是 0。
        const noSash = createGrid<string>(null);
        noSash.restore(snapshot, (ref) => ({ref}));
        log("P5e 新网格忘给 sashSize：sashSizes", noSash.layout({width: 1000, height: 600}).sashSizes);
    });
});

    it("按比例分的目标：嵌套与并入呈现是否相同", () => {
        const nested = layoutOf({kind: "branch", id: "root", orientation: "horizontal", children: [
            {kind: "leaf", id: "a", ref: "a", size: {width: 300, height: 0}},
            {kind: "branch", id: "s", orientation: "horizontal", size: {width: 300, height: 0}, children: [
                {kind: "leaf", id: "n", ref: "n", size: {width: 150, height: 0}},
                {kind: "leaf", id: "t", ref: "t", size: {width: 150, height: 0}},
            ]},
            {kind: "leaf", id: "b", ref: "b", size: {width: 300, height: 0}},
        ]});
        const merged = layoutOf({kind: "branch", id: "root", orientation: "horizontal", children: [
            {kind: "leaf", id: "a", ref: "a", size: {width: 300, height: 0}},
            {kind: "leaf", id: "n", ref: "n", size: {width: 150, height: 0}},
            {kind: "leaf", id: "t", ref: "t", size: {width: 150, height: 0}},
            {kind: "leaf", id: "b", ref: "b", size: {width: 300, height: 0}},
        ]});
        log("P7a 按比例分（1204 宽、缝 4）", {nested: widths(nested.sizes, ["a", "n", "t", "b"]), merged: widths(merged.sizes, ["a", "n", "t", "b"])});
    });
    it("全 0 意图（Lab live-panels 的建法）：嵌套是 1/4+1/4+1/2，并入按公式是 0.5/0.5/0", () => {
        const nested = layoutOf({kind: "branch", id: "col", orientation: "vertical", children: [
            {kind: "branch", id: "s", orientation: "vertical", size: {width: 0, height: 0}, children: [
                {kind: "leaf", id: "c", ref: "c", size: {width: 0, height: 0.5}},
                {kind: "leaf", id: "n", ref: "n", size: {width: 0, height: 0.5}},
            ]},
            {kind: "leaf", id: "d", ref: "d"},
        ]});
        const mergedFormula = layoutOf({kind: "branch", id: "col", orientation: "vertical", children: [
            {kind: "leaf", id: "c", ref: "c", size: {width: 0, height: 0.5}},
            {kind: "leaf", id: "n", ref: "n", size: {width: 0, height: 0.5}},
            {kind: "leaf", id: "d", ref: "d"},
        ]});
        const mergedZero = layoutOf({kind: "branch", id: "col", orientation: "vertical", children: [
            {kind: "leaf", id: "c", ref: "c"}, {kind: "leaf", id: "n", ref: "n"}, {kind: "leaf", id: "d", ref: "d"},
        ]});
        const h = (sizes: Record<string, {width: number; height: number}>) => Object.fromEntries(["c", "n", "d"].map((id) => [id, Math.round(sizes[id]!.height * 100) / 100]));
        log("P7b 全 0 意图（600 高）", {nested: h(nested.sizes), mergedUnit1: h(mergedFormula.sizes), mergedZero: h(mergedZero.sizes)});
    });
    it("约束：嵌套分支的最小值是两孩子之和，并入后各算各的；窄容器下压缩是否不同", () => {
        const kids = (prefix: string) => [
            {kind: "leaf" as const, id: `${prefix}n`, ref: "n", size: {width: 150, height: 0}, minimumSize: {width: 120, height: 0}},
            {kind: "leaf" as const, id: `${prefix}t`, ref: "t", size: {width: 150, height: 0}, minimumSize: {width: 120, height: 0}},
