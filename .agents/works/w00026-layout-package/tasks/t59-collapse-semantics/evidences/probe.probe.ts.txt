type Ext = {width: number; height: number};
const U = Number.POSITIVE_INFINITY;
const W = (v: number) => ({width: v, height: 0});

function log(name: string, value: unknown): void {
    console.log(`[probe] ${name}: ${JSON.stringify(value)}`);
}

function widths(grid: any, ext: Ext, ids: string[]): number[] {
    const sizes = grid.layout(ext).sizes;
    return ids.map((id) => Math.round((sizes[id]?.width ?? NaN) * 100) / 100);
}

function session(grid: any, ext: Ext, branchId: string, index: number, source: "keyboard" | "pointer", axis: "width" | "height" = "width") {
    return createGridGestureSession({contextKey: "p", revision: 1, root: grid.root(), extent: ext, layout: grid.layout(ext), sashSize: 4, sessionId: "p", source, sashes: [{branchId, index, axis}]});
}

function commit(grid: any, gesture: any): any {
    const finished = gesture.finish();
    if (finished === null) return {none: true};
    return grid.resizeBranches(finished.changes);
}

const POLICY = {collapsedSize: 0, collapseThreshold: 24, expandThreshold: 24};

describe("P1 第 35 条：意图 400、策略 restoreSize 333", () => {
    const build = () => createGrid({
        kind: "branch", id: "root", orientation: "horizontal", children: [
            {kind: "leaf", id: "a", ref: "a", size: W(600)},
            {kind: "leaf", id: "c", ref: "c", size: W(400), sizing: "fixed", minimumSize: W(100), collapse: {...POLICY, collapsed: true, restoreSize: 333}},
        ],
    }, {sashSize: 4});
    const EXT = {width: 1004, height: 400};

    it("setCollapsed(false) 回到意图 400", () => {
        const grid = build();

        const w = widths(grid, EXT, ["a", "c"]);
        log("P1 setCollapsed(false)", w);
        expect(w[1]).toBe(400);
    });

    it("分隔线上 Enter 回到 restoreSize 333", () => {

        const r = commit(grid, s);

        log("P1 Enter committed", {ok: r.ok, widths: w, intentC: grid.find("c").size.width});
        expect(w[1]).toBe(333);
    });
});

/** ide 的主行：[左条, 左区, 中间, 右区, 右条]，条没有最小与最大（第 45 条的原始写法）或钉死（示例现在的绕法）。 */
function ideRow(pinned: boolean) {
    const stripe = (id: string) => pinned
        ? {kind: "leaf", id, ref: id, size: W(28), sizing: "fixed", minimumSize: W(28), maximumSize: {width: 28, height: U}}
        : {kind: "leaf", id, ref: id, size: W(28), sizing: "fixed"};
    const zone = (id: string, size: number, collapsed: boolean) => ({kind: "leaf", id, ref: id, size: W(size), sizing: "fixed", minimumSize: W(160), collapse: {...POLICY, collapsed, restoreSize: size}});
    return createGrid({
        kind: "branch", id: "main", orientation: "horizontal", children: [
            stripe("ls"), zone("left", 260, false),
            {kind: "leaf", id: "center", ref: "center", size: {width: 1, height: 1}, minimumSize: {width: 300, height: 200}},
            zone("right", 240, false), stripe("rs"),
        ],
    }, {sashSize: 4});
}
const IDE_IDS = ["ls", "left", "center", "right", "rs"];

describe("P3 第 45 条：级联把 28px 的条挤到 0", () => {
    const EXT = {width: 1400, height: 800};
    for (const pinned of [false, true]) {
        it(`拖 中间|右区 往右 900px，条${pinned ? "钉死" : "没有最小最大"}`, () => {
            const grid = ideRow(pinned);
            log(`P3 pinned=${pinned} before`, widths(grid, EXT, IDE_IDS));
            const s = session(grid, EXT, "main", 2, "pointer");
            s.update({x: 900, y: 0});
            const r = commit(grid, s);
            const after = widths(grid, EXT, IDE_IDS);
            log(`P3 pinned=${pinned} after`, {ok: r.ok, after, rightCollapsed: grid.find("right").collapse.collapsed});
            if (!pinned) expect(after[4]).toBe(0);
            else expect(after[4]).toBe(28);
        });
    }
});

/**
 * P9 候选规则的模拟（直接改节点意图，只为量数，不是产品写法）：
 * 规则 R1：当前意图（翻过收起标记之后）就能呈现目标时，整支意图不改。
 * 规则 R2：否则照旧反解，之后落账后展开着的按比例分的孩子按同一倍数缩放，使它们的意图之和等于落账前之和
 *          （只在落账后还有收起着的按比例分的孩子时做）。
 */
function rescaleExpandedWeights(grid: any, branchId: string, oldIntents: Record<string, number>): void {
    const branch = grid.find(branchId);
    const weights = branch.children.filter((child: any) => child.sizing !== "fixed");
    const collapsedWeights = weights.filter((child: any) => child.collapse?.collapsed === true);
    if (collapsedWeights.length === 0) return;
    const expanded = weights.filter((child: any) => child.collapse?.collapsed !== true);
    const oldSum = expanded.reduce((sum: number, child: any) => sum + oldIntents[child.id]!, 0);
    const newSum = expanded.reduce((sum: number, child: any) => sum + child.size.width, 0);
    if (oldSum <= 0 || newSum <= 0) return;
    for (const child of expanded) child.size.width *= oldSum / newSum;
}

    it("两块：拖到收起 b，R2 之后展开回到 500", () => {
        const ROW = {width: 1004, height: 400};
        const grid = createGrid({
            kind: "branch", id: "root", orientation: "horizontal", children: [
                {kind: "leaf", id: "a", ref: "a", size: W(500), minimumSize: W(100)},
                {kind: "leaf", id: "b", ref: "b", size: W(500), minimumSize: W(100), collapse: {...POLICY, collapsed: false, restoreSize: 123}},
            ],
        }, {sashSize: 4});
        const old = {a: 500, b: 500};
        const s = session(grid, ROW, "root", 0, "pointer");
        s.update({x: 600, y: 0});
        commit(grid, s);
        const beforeRescale = widths(grid, ROW, ["a", "b"]);
        rescaleExpandedWeights(grid, "root", old);
        log("P9 two: presentation unchanged by R2", {beforeRescale, after: widths(grid, ROW, ["a", "b"]), intents: [grid.find("a").size.width, grid.find("b").size.width]});
        grid.setCollapsed("b", false);
        log("P9 two: expand by share", widths(grid, ROW, ["a", "b"]));

    it("三块：拖 b|c 把 c 收起（b 吸收），R2 之后展开 c", () => {
        const EXT = {width: 1208, height: 400};
        const grid = createGrid({

                {kind: "leaf", id: "a", ref: "a", size: W(300), minimumSize: W(80)},
                {kind: "leaf", id: "b", ref: "b", size: W(300), minimumSize: W(80)},
